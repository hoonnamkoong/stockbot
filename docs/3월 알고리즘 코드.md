## 1. 룰 기반 스코어링 엔진 (`src/strategy/engine.py`)
기본적인 수급과 추세를 점수로 환산하고 1차 시그널을 결정하는 로직입니다.

```python
# 점수 산정 로직
def calculate_score(self, stock_data):
    trend_score = 0
    p_change = float(str(stock_data.get('change_rate', '0')).replace('%', ''))
    
    # [Trend] 주가 등락률 기준
    if p_change > 2.0: trend_score += 20
    if p_change > 5.0: trend_score += 10
    
    # [Supply] 외국인 보유 비중 기준
    supply_score = 0
    frg = float(str(stock_data.get('foreign_rate', '0')).replace('%', ''))
    if frg > 0: supply_score += 10
    if frg > 5: supply_score += 10
        
    return trend_score + supply_score, p_change

# 시그널 판정 및 포트폴리오 오버라이드
def get_signal(self, score, p_change, in_portfolio=False, profit_rate=0.0):
    signal = "HOLD"
    
    # 기초 시그널 (임계치 정적 룰)
    if score >= 30: # buy_strong threshold
        signal = "BUY_STRONG"
    elif score >= 20: # buy threshold
        signal = "BUY"
    elif p_change < -3.0: # sell threshold
        signal = "SELL"
        
    # 포트폴리오 보유 시 수익률 기반 강제 오버라이드 (V12 핵심 로직)
    if in_portfolio:
        if profit_rate >= 10.0: # 10% 이상 익절 구간
            # 추세가 살아있으면 홀딩(Ride the Winner), 아니면 익절
            if p_change > 2.0 or signal == "BUY_STRONG":
                signal = "HOLD"
            else:
                signal = "SELL"
        elif profit_rate <= -7.0: # -7% 손절 구간
            signal = "SELL" # 무조건 손절
            
    return signal
```

## 3. 커뮤니티 데이터 퀄리티 필터 (`analyzer.py`)
노이즈를 제거하고 유의미한 정보가 포함된 게시물을 선별하는 가중치 로직입니다.

```python
# 게시물 점수 산정 (Deep Dive Weighted Scoring V7.5)
for p in latest_posts:
    # 1. 기초 점점수 (조회수 + 좋아요 가중치)
    views = int(str(p.get('views', '0')).replace(',', ''))
    likes = int(p.get('likes', '0'))
    score = views + (likes * 30)

    # 2. 본문 내용 분석 및 패널티/보너스
    body = p.get('body', '')
    if body:
        # 패널티: 2줄 이하 또는 50자 미만의 짧은 글 (80% 삭감)
        if len(body) < 50 or len(body.split('\n')) <= 2:
            score *= 0.2
        
        # 보너스: 분석적 키워드 포함 시 (+2000 가산점)
        prediction_keywords = ['목표', '예상', '전망', '분석', '이유']
        if any(kw in body for kw in prediction_keywords):
            score += 2000
    
    p['score'] = score
```
